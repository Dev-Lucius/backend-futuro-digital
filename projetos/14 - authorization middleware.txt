// /server.js

require('dotenv').config();
const express = require('express');
const validator = require('express-openapi-validator');
const jwt = require('jsonwebtoken');

const server = express();
const port = process.env.EXPRESS_PORT;

const rolesMiddleware = (roles) => (request, response, next) => {
	if (!roles.includes(request.jwtpayload.role)) {
		response.status(403).json();
		return;
	}
	next();
};

server.use(express.json());

server.use(
	validator.middleware({
		apiSpec: 'api.yaml',
		validateRequests: true,
		validateResponses: false,
		validateSecurity: {
			handlers: {
				bearer: (request, scopes, schema) => {
					request.jwtpayload = jwt.verify(request.headers.authorization.replace('Bearer ', ''), process.env.JWT_SECRET);
					return(true);
				}
			}
		}
	})
);

server.post('/autenticacao', (request, response) => {
	let role = 0;
	if (request.body.usuario == 'fulano') {
		role = 1;
	}
	if (request.body.usuario == 'beltrano') {
		role = 2;
	}
	const token = jwt.sign({ sub: request.body.usuario, role: role }, process.env.JWT_SECRET, { expiresIn: '1m' });
	console.log(token);
	response.status(200).json({ token: token });
});

server.get('/publico', (request, response) => {
	response.status(200).json('resposta');
});

server.get('/privado', (request, response) => {
	response.status(200).json('resposta');
});

//server.get('/privado/administracao', (request, response) => {

//	{	response.status(403).json();
//		return;
//	}
//	response.status(200).json('resposta');
//});

server.get('/privado/administracao', rolesMiddleware([ 2, 5 ]), (request, response) => {
	response.status(200).json('resposta');
});

server.use((error, request, response, next) => {
	console.log(error.stack);
	response.status(error.status || 500).json(error.message);
});

server.listen(port, () => {
	console.log(`Server listening at http://localhost:${port}`);
});
