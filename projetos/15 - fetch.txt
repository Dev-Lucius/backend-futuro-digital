// /server.js

require('dotenv').config();
const express = require('express');
const validator = require('express-openapi-validator');

const server = express();
const port = process.env.EXPRESS_PORT;
const url = 'https://jsonplaceholder.typicode.com/posts';

server.use(express.json());

server.use(validator.middleware({ apiSpec: 'api.yaml', validateRequests: true, validateResponses: false }));

server.get('/get', async (request, response) => {
	try {
		const page = 1;
		const limit = 5;
		const fetch_response = await fetch(`${url}?_page=${page}&_limit=${limit}`, {
			method: 'GET',
			headers: {
				'Accept': 'application/json'
			}
		});
		if (!fetch_response.ok) {
			response.status(500).json(fetch_response.status);
			return;
		}
		const data = await fetch_response.json();
		console.log(data);
		response.status(200).json('resposta');
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.get('/post', async (request, response) => {
	try {
		const fetch_response = await fetch(url, {
			method: 'POST',
			headers: {
				'Content-Type': 'application/json',
				'Accept': 'application/json'
			},
			body: JSON.stringify({
				userId: 1,
				title: 'titulo',
				body: 'conteudo'
			})
		});
		if (!fetch_response.ok) {
			response.status(500).json(fetch_response.status);
			return;
		}
		const data = await fetch_response.json();
		console.log(data);
		response.status(200).json('resposta');
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.get('/put', async (request, response) => {
	try {
		const id = 1;
		const fetch_response = await fetch(`${url}/${id}`, {
			method: 'PUT',
			headers: {
				'Content-Type': 'application/json',
				'Accept': 'application/json'
			},
			body: JSON.stringify({
				id: id,
				userId: 1,
				title: 'titulo',
				body: 'conteudo'
			})
		});
		if (!fetch_response.ok) {
			response.status(500).json(fetch_response.status);
			return;
		}
		const data = await fetch_response.json();
		console.log(data);
		response.status(200).json('resposta');
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.get('/delete', async (request, response) => {
	try {
		const id = 1;
		const fetch_response = await fetch(`${url}/${id}`, {
			method: 'DELETE'
		});
		if (!fetch_response.ok) {
			response.status(500).json(fetch_response.status);
			return;
		}
		const data = await fetch_response.json();
		console.log(data);
		response.status(200).json('resposta');
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.use((error, request, response, next) => {
	console.log(error.stack);
	response.status(error.status || 500).json(error.message);
});

server.listen(port, () => {
	console.log(`Server listening at http://localhost:${port}`);
});
