// /server.js

require('dotenv').config();
const express = require('express');
const validator = require('express-openapi-validator');
const db = require('pg');

const server = express();
const port = process.env.EXPRESS_PORT;
const pool = new db.Pool({
	host: process.env.DB_HOST,
	port: process.env.DB_PORT,
	database: process.env.DB_DATABASE,
	user: process.env.DB_USER,
	password: process.env.DB_PASSWORD
});

server.use(express.json());

server.use(validator.middleware({ apiSpec: 'api.yaml', validateRequests: true, validateResponses: false }));

server.get('/', async (request, response) => {
	try {
		const result = await pool.query('SELECT * FROM pessoa');
		console.log(result);
		response.status(200).json(result.rows);
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.post('/', async (request, response) => {
	try {
		const result = await pool.query('INSERT INTO pessoa (nome, nascimento, genero) VALUES ($1, $2, $3) RETURNING *', [ request.body.nome, request.body.nascimento, request.body.genero ]);
		console.log(result);
		response.status(201).json(result.rows[0]);
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.get('/:codigo', async (request, response) => {
	try {
		const result = await pool.query('SELECT * FROM pessoa WHERE codigo = $1', [ request.params.codigo ]);
		console.log(result);
		if (result.rows.length == 0) {
			response.status(404).json();
			return;
		}
		response.status(200).json(result.rows[0]);
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.put('/:codigo', async (request, response) => {
	try {
		const result = await pool.query('UPDATE pessoa SET nome = $1, nascimento = $2, genero = $3 WHERE codigo = $4 RETURNING *', [ request.body.nome, request.body.nascimento, request.body.genero, request.params.codigo ]);
		console.log(result);
		if (result.rows.length == 0) {
			response.status(404).json();
			return;
		}
		response.status(200).json(result.rows[0]);
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.delete('/:codigo', async (request, response) => {
	try {
		const result = await pool.query('DELETE FROM pessoa WHERE codigo = $1 RETURNING *', [ request.params.codigo ]);
		console.log(result);
		if (result.rows.length == 0) {
			response.status(404).json();
			return;
		}
		response.status(204).json();
	} catch (error) {
		console.log(error.stack);
		response.status(500).json(error.message);
	}
});

server.use((error, request, response, next) => {
	console.log(error.stack);
	response.status(error.status || 500).json(error.message);
});

server.listen(port, () => {
	console.log(`Server listening at http://localhost:${port}`);
});
